
#pragma once

	class <$NAME> : public HexEngine::IGameExtension
	{
	public:
		// Lifecycle
		//
		// Standalone launcher: OnRegisterClasses -> OnLoadGameWorld -> OnCreateGame -> (loop) -> OnStopGame -> OnShutdown

		//
		virtual void OnCreateGame() override;

		virtual void OnStopGame() override;

		virtual void OnLoadGameWorld() override;

		virtual void OnRegisterClasses() override;

		virtual void OnUpdate(float frameTime) override;

		virtual void OnFixedUpdate(float frameTime) override;

		virtual void OnShutdown() override;

		virtual void OnDebugGUI() override;

		virtual void OnGUI() override;

		virtual void OnDebugRender() override;

		virtual void OnResize(int32_t width, int32_t height) override {}

		virtual std::string GetGameName() const override { return "<$NAME>"; }

	private:
		std::shared_ptr<HexEngine::Scene> _scene;

		// The camera controller this game attached in OnCreateGame, so OnStopGame
		// can take it off again. In the editor the main camera persists between
		// play sessions and components added during play are not rolled back.
		HexEngine::Entity* _cameraEntity = nullptr;
		bool _addedCameraController = false;
	};

	inline <$NAME>* g_pGame = nullptr;
